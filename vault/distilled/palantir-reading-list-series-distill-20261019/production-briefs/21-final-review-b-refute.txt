const verifyPrompt = (label, r) => `${CONTEXT}
A reviewer (${label}) reported the findings below on the final mp4s. Your job is to try to REFUTE each must and should
finding by looking at the actual frames yourself (dense frames via sheets.py into ${S}/review-b/verify-${label}/; at
most ~200 frames in total). A finding is real only if the defect is visible in the frames at (or within a second of) the
stated time and a first-time viewer would notice it. Also say whether the severity is right. Skip could-findings (do not
include them). Default to real=false when you cannot see it.

Findings (index: finding):
${r.findings.map((f, i) => ({ i, f })).filter(x => x.f.severity !== 'could').map(x => `${x.i}: ${JSON.stringify(x.f)}`).join('\n')}`

const passes = [
  { label: 'ep00-02', prompt: episodePass(['ep00', 'ep01', 'ep02']) },
  { label: 'ep03-05', prompt: episodePass(['ep03', 'ep04', 'ep05']) },
  { label: 'series', prompt: seriesPass },
]

const out = await pipeline(passes,
  p => agent(p.prompt, { label: `review:${p.label}`, phase: 'Review', schema: FINDINGS }),
  (r, p) => {
    if (!r) return null
    const toCheck = r.findings.filter(f => f.severity !== 'could')
    if (!toCheck.length) return { label: p.label, review: r, verdicts: [] }
    return agent(verifyPrompt(p.label, r), { label: `verify:${p.label}`, phase: 'Verify', schema: VERDICTS, model: 'sonnet' })
      .then(v => ({ label: p.label, review: r, verdicts: v ? v.results : null }))
  })
return out
